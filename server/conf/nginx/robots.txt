# Local development only. Deployed portals don't serve this file: Camino's nginx answers
# /robots.txt from /var/www/robots.txt, which is Camino's robots.txt.default unless the

#

# directory's nginx.conf replaces every occurrence with that server block's $server_name
# (sub_filter), so the Sitemap: lines always name the host nginx is configured for.
